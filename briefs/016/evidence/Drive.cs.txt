    static System.Collections.IEnumerator Frames(int n) { for (int i = 0; i < n; i++) yield return null; }
    static System.Collections.IEnumerator Press(string key)
    {

    static System.Collections.IEnumerator Mouse(float x, float y, bool mid) { MouseAt(x, y, mid); yield return Frames(3); }
    static System.Collections.IEnumerator Shot(System.Func<string> take) { while (HudToolkit.ShotBusy) yield return null; L(take()); yield return Frames(2); while (HudToolkit.ShotBusy) yield return null; }
    static System.Collections.IEnumerator RouteCo()
    {
        var h = H; Application.runInBackground = true; h.PeekLatch = false;

        L("start " + Start());
        yield return Press("E");
        L("loop " + Loop()); L("watch " + ToWatch());
        KeyDown("LeftAlt"); yield return Frames(3);
        yield return Mouse(800, 500, true); L($"alt+middle down: drag {h.TurnDragging} peeking {h.Peeking}");
        yield return Mouse(1000, 500, true); L($"dragged 200 px: yaw {h.Yaw:0.#}");
        yield return Mouse(1000, 500, false); UnityEngine.InputSystem.InputSystem.QueueStateEvent(UnityEngine.InputSystem.Keyboard.current, new UnityEngine.InputSystem.LowLevel.KeyboardState());

        yield return Mouse(-50, -50, false);
        L("takedown armed " + ToTakedown());
        L(FrameSet("a"));
        for (int k = 0; k < 4; k++) { int kk = k; yield return Shot(() => $"a yaw {30 + 90 * kk}: " + RestShot(kk, $"a-{30 + 90 * kk:000}") + " · " + Who().Replace("\n", " | ")); }

        if (h.Picked >= 0) h.Cancel(); Turn(0); L(FrameSet("b"));
        for (int k = 0; k < 4; k++) { int kk = k; yield return Shot(() => $"b yaw {30 + 90 * kk}: " + RestShot(kk, $"b-{30 + 90 * kk:000}") + " · " + Who().Replace("\n", " | ")); }

        h.Select(OpId.Scout, false); yield return Shot(() => "cameos yaw 120: " + RestShot(1, "edge-cameos-120"));
        h.Select(OpId.Enforcer, false); Turn(0);
        L("card " + OpenCard()); L("knockout " + KnockOut());
        yield return Press("Q");
        L("enter " + Enter());
        yield return Press("Q");
        L("finish " + Finish());
        int g = 0; while (h.Ending.Length == 0 && g++ < 600) { h.RunTo(h.Game.Tick + 10); yield return null; }
        L($"ending '{h.Ending}' yaw {h.Yaw:0} session {h.SessionPath}");

    /// <summary>Sweeps the screen for a ground point whose preview is `state` for the selected agent, leaves the cursor there.</summary>
    public static string Find(string state) => FindW(state, 0);
    public static string FindW(string state, int minWalk)
    {
        var h = H; var cam = h.View; Vector2 best = default; bool found = false; int tried = 0;
        for (float fy = 0.25f; fy <= 0.85f && !found; fy += 0.05f)
            for (float fx = 0.2f; fx <= 0.85f && !found; fx += 0.04f)
            {
                var sp = new Vector2(fx * cam.pixelWidth, (1 - fy) * cam.pixelHeight); tried++;
                h.PreviewAt(sp); var mv = h.MoveNow;
                if (mv != null && mv.Refused.Length == 0 && mv.State == state && mv.WalkS >= minWalk && (state != "seen" || mv.Glow.Count > 0)) { best = sp; found = true; }
            }
        if (!found) return $"no {state} point in {tried}";
        h.PreviewAt(best); var m = h.MoveNow;
        return $"{state} at {best} walk {m.WalkS} s seenKind '{m.SeenKind}' glow {m.Glow.Count} unwatched {m.Unwatched.Count}";
    }
