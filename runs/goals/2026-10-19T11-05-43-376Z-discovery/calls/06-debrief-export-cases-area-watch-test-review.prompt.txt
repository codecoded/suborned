Any run can be replayed exactly from its starting state and the player's orders.

        /// <summary>Play a start state and its orders (sorted by tick, stable) to `until` or the outcome.</summary>
        public static Game Play(Inputs i, Start start, IReadOnlyList<Order> orders, int until)
        {
            var g = New(i, start);
            if (until <= 0) until = Game.T(i.SiteData.Over); // a script without `until` plays to the close mission.json gives
            var list = new List<(int idx, Order o)>(); for (int k = 0; k < orders.Count; k++) list.Add((k, orders[k]));
            list.Sort((a, b) => a.o.Tick != b.o.Tick ? a.o.Tick.CompareTo(b.o.Tick) : a.idx.CompareTo(b.idx));
            foreach (var (_, o) in list)
            {
                if (o.Tick > until) break; // orders past `until` are not played (Redline I2)
                g.RunTo(o.Tick);
                if (g.Over && g.Tick != o.Tick) break; // the mission ended before this order's tick: nothing later was given
                if (g.Tick == o.Tick) g.Issue(o);      // at the tick it ended, the played run took (and refused) it: so does the replay
            }
            g.RunTo(until); g.Finish();
            return g;
        }
        /// <summary>Replay a finished game from its own start state and order log.</summary>
        public static Game Replay(Inputs i, Game played)
        {
            var orders = new List<Order>(); foreach (var (o, _, _, _) in played.Orders) orders.Add(o);
            return Play(i, played.Start, orders, played.Tick);
        }

    /// <summary>The seeded streams (rulebook streams): FNV-1a 32 and mulberry32, integer draws; the same as contracts/rules/rulebook.ref.ts.</summary>
    public static class Streams
    {
        public static uint Fnv1a32(string key) { unchecked { uint h = 0x811c9dc5; foreach (var c in key) { h ^= c; h *= 0x01000193; } return h; } }
        public sealed class Mulberry
        {
            uint s;
            public Mulberry(string key) { s = Fnv1a32(key); if (s == 0) s = 1; }
            public uint Next()
            {
                unchecked
                {
                    s += 0x6d2b79f5; uint t = s;
                    t = (t ^ (t >> 15)) * (t | 1);
                    t ^= t + (t ^ (t >> 7)) * (t | 61);
                    return t ^ (t >> 14);
                }
            }
        }
        /// <summary>x = floor(u × total ÷ 2^32); walk the items in order subtracting each weight.</summary>
        public static T Draw<T>(string key, IReadOnlyList<(T item, int weight)> items)
        {
            if (items.Count == 0) throw new ArgumentException("nothing to draw");
            long total = 0; foreach (var i in items) { if (i.weight <= 0) throw new ArgumentException("weights are positive"); total += i.weight; }
            long x = (long)(((ulong)new Mulberry(key).Next() * (ulong)total) >> 32);
            foreach (var i in items) { x -= i.weight; if (x < 0) return i.item; }
            return items[items.Count - 1].item;
        }
    }

    static int Play(Engine.Inputs inputs, string root, string what, bool json)
    {
        Game g; List<(bool ok, string what)> checks = new List<(bool, string)>(); string id; string record = "";
        var sc = Scenarios.Find(what);
        if (sc != null) { var run = Scenarios.Run(inputs, sc); g = run.G; checks = run.Checks; id = sc.Id; }
        else
        {
            var path = File.Exists(what) ? what : System.IO.Path.Combine(root, what);
            var script = Script.Parse(File.ReadAllText(path));
            g = Engine.Play(inputs, script.Start, script.Orders, script.Until); id = System.IO.Path.GetFileName(path);
            if (script.Expect != null) { record = script.Verify(g, inputs.Hash); checks.Add((record.Length == 0, record.Length == 0 ? "the replay matches its record (inputs, engine, start, every checkpoint, outcome)" : "record: " + record)); }
        }
        var replay = Engine.Replay(inputs, g);
        bool same = Same(g, replay);
        bool passed = checks.TrueForAll(c => c.ok);
        if (json) { Console.WriteLine(Report.Json(id, g, checks, same)); }
        else
        {
            Console.WriteLine($"{id}: {g.Result}{(g.FailCode.Length > 0 ? " (" + g.FailCode + ")" : "")} at {Report.Clock(g.Tick)}  cash {g.Cash}  trail {g.Trail}  alert {g.Alert}  shots {g.Gunfire}");
            foreach (var l in g.Log) Console.WriteLine($"  {Report.Clock(l.Tick)}  {l.Code,-28} {l.Text}");
            foreach (var c in checks) Console.WriteLine($"  [{(c.ok ? "ok" : "FAIL")}] {c.what}");
            if (g.Result == Outcome.Success) { var w = new JWriter(); Debrief.Of(g).Write(w); Console.WriteLine($"  debrief {w}"); }
            if (g.Cause != null) Console.WriteLine($"  cause {g.Cause.Json()}");
            Console.WriteLine($"  start {Hash.Hex(g.StartHash)}  final {Hash.Hex(g.Checkpoints[g.Checkpoints.Count - 1].hash)}  checkpoints {g.Checkpoints.Count}  replay {(same ? "identical" : "DIFFERS")}");
        }
        return passed && same ? 0 : 1;
    }

    /// <summary>Every scenario's and recorded route script's checkpoints, with the start state's text given (default: the inputs hash).</summary>
    static string Hashes(Engine.Inputs i, string root, string? startText)
    {
        string st = startText ?? i.Hash.ToString(CultureInfo.InvariantCulture);
        var w = new JWriter().Obj().Prop("inputs", i.Hash.ToString(CultureInfo.InvariantCulture)).Prop("startText", st);
        void put(string id, Game g) { w.OpenArr(id); foreach (var (t, h) in g.Checkpoints) w.Str(t.ToString(CultureInfo.InvariantCulture) + ":" + Hash.Hex(h)); w.Str("result:" + g.Result + ":" + g.FailCode); w.EndArr(); }
        foreach (var sc in Scenarios.All)
        {
            var r = new Run(new Game(i.Rules, i.SiteData, sc.Start, st, i.Crowd)); sc.Body(r); r.G.Finish(); put(sc.Id, r.G);
        }
        var files = new List<string>();
        foreach (var d in new[] { "contracts/missions/handover/routes", "briefs/011/motive-fights" })
            if (Directory.Exists(System.IO.Path.Combine(root, d))) foreach (var f in Directory.GetFiles(System.IO.Path.Combine(root, d), "*.json")) files.Add(f);
        files.Sort(StringComparer.Ordinal);
        foreach (var f in files)
        {
            var txt = File.ReadAllText(f); if (!txt.Contains("suborned.sim.script/1")) continue;
            var sc = Script.Parse(txt); var g = new Game(i.Rules, i.SiteData, sc.Start, st, i.Crowd);
            // Engine.Play with the given start text
            var list = new List<(int idx, Order o)>(); for (int k = 0; k < sc.Orders.Count; k++) list.Add((k, sc.Orders[k]));
            list.Sort((a, b) => a.o.Tick != b.o.Tick ? a.o.Tick.CompareTo(b.o.Tick) : a.idx.CompareTo(b.idx));
            int until = sc.Until <= 0 ? Game.T(i.SiteData.Over) : sc.Until;
            foreach (var (_, o) in list) { if (o.Tick > until) break; g.RunTo(o.Tick); if (g.Over && g.Tick != o.Tick) break; if (g.Tick == o.Tick) g.Issue(o); }
            g.RunTo(until); g.Finish();
            put(System.IO.Path.GetRelativePath(root, f), g);
        }
        return w.EndObj().ToString();
    }

    /// <summary>The fans of an idle run at a tick, as JSON, and what building and reading them cost (Keel, cones 2026-10-16).</summary>
    static int Fans(Engine.Inputs i, int tick, bool crowd)
    {
        var g = Engine.New(i); g.RunTo(tick);
        var w = new JWriter().Obj().Prop("tick", g.Tick).Prop("crowd", crowd).OpenArr("fans");
        foreach (var id in g.WatcherIds())
        {
            var f = g.ConeFan(id, crowd); if (f == null) continue;
            w.Obj().Prop("watcher", f.Watcher).Prop("kind", f.Kind).Prop("live", f.Live).Prop("reach", f.Reach).Prop("points", f.Points.Length);
            w.Key("poly").Raw("[" + string.Join(",", Array.ConvertAll(f.Points, p => $"[{p.X},{p.Y}]")) + "]");
            w.Key("stop").Raw("[" + string.Join(",", Array.ConvertAll(f.Stop, x => "\"" + x + "\"")) + "]");
            w.EndObj();
        }
        w.EndArr();
        // cost: a cold build of every fan (a fresh game each time, so nothing is cached), and a warm read (cached by pose)
        double coldMs = 0; for (int k = 0; k < 50; k++) { var h = Engine.New(i); h.RunTo(tick); var c = Stopwatch.StartNew(); foreach (var id in h.WatcherIds()) h.ConeFan(id, crowd); c.Stop(); coldMs += c.Elapsed.TotalMilliseconds; }
        var warm = Stopwatch.StartNew(); for (int k = 0; k < 10000; k++) foreach (var id in g.WatcherIds()) g.ConeFan(id, crowd); warm.Stop();
        w.Key("coldMsAllFans").Raw((coldMs / 50).ToString("0.0000", CultureInfo.InvariantCulture)).Key("warmUsAllFans").Raw((warm.Elapsed.TotalMilliseconds * 1000 / 10000).ToString("0.000", CultureInfo.InvariantCulture));
        Console.WriteLine(w.EndObj().ToString());
        return 0;
    }

    static bool Same(Game a, Game b)
    {
        if (a.Checkpoints.Count != b.Checkpoints.Count) return false;
        for (int i = 0; i < a.Checkpoints.Count; i++) if (a.Checkpoints[i] != b.Checkpoints[i]) return false;
        return true;
    }

    static int Perf(Engine.Inputs inputs, int runs, string id)
    {
        var sc = Scenarios.Find(id) ?? throw new ArgumentException($"no scenario {id}");
        Scenarios.Run(inputs, sc); // warm
        var sw = Stopwatch.StartNew(); long ticks = 0;
        for (int i = 0; i < runs; i++) ticks += Scenarios.Run(inputs, sc).G.Tick;
        sw.Stop();
        double simS = ticks / 10.0, wallS = sw.Elapsed.TotalSeconds;
        Console.WriteLine(FormattableString.Invariant($"{sc.Id} x{runs}: {simS:0} sim-s in {wallS:0.000} wall-s = {simS / wallS:0} sim-s per wall-s ({wallS * 1000 / runs:0.0} ms a run)"));
        return 0;
    }

    public static class Report
    {
        public static string Clock(int tick) { int s = tick / Game.Hz; return $"{s / 60}:{(s % 60).ToString("00", CultureInfo.InvariantCulture)}.{tick % Game.Hz}"; }
        public static string Json(string id, Game g, List<(bool ok, string what)> checks, bool replayIdentical)
        {
            var w = new JWriter().Obj().Prop("id", id).Prop("outcome", g.Result.ToString().ToLowerInvariant()).Prop("fail", g.FailCode).Prop("tick", g.Tick)
                .Prop("cash", g.Cash).Prop("trail", g.Trail).Prop("alert", g.Alert).Prop("gunfire", g.Gunfire).Prop("standing", g.Standing)
                .Prop("startHash", Hash.Hex(g.StartHash)).Prop("finalHash", Hash.Hex(g.Checkpoints[g.Checkpoints.Count - 1].hash)).Prop("replayIdentical", replayIdentical);
            if (g.Cause != null) { w.Key("cause"); g.Cause.Write(w); } // the fail camera's culprit (FailCause.cs): written only when the run failed or withdrew
            w.OpenArr("checkpoints"); foreach (var (t, h) in g.Checkpoints) w.Obj().Prop("tick", t).Prop("hash", Hash.Hex(h)).EndObj(); w.EndArr();
            w.OpenArr("checks"); foreach (var c in checks) w.Obj().Prop("ok", c.ok).Prop("what", c.what).EndObj(); w.EndArr();
            w.OpenArr("orders"); foreach (var (o, ok, code, why) in g.Orders) { w.Obj().Key("order").Raw(o.ToString()).Prop("ok", ok); if (!ok) w.Prop("code", code).Prop("why", why); w.EndObj(); } w.EndArr();
            w.OpenArr("ledger"); foreach (var (t, what, d, now, cause) in g.Ledger) w.Obj().Prop("tick", t).Prop("what", what).Prop("delta", d).Prop("now", now).Prop("cause", cause).EndObj(); w.EndArr();
            w.OpenArr("log"); foreach (var l in g.Log) w.Obj().Prop("tick", l.Tick).Prop("code", l.Code).Prop("text", l.Text).EndObj(); w.EndArr();
            if (g.Result == Outcome.Success) { w.Key("debrief"); var d = new JWriter(); Debrief.Of(g).Write(d); w.Raw(d.ToString()); }
            return w.EndObj().ToString();
        }

    static readonly Engine.Inputs I = Repo.Inputs;
    static readonly string Root = Engine.FindRoot(AppContext.BaseDirectory);
    static readonly string[] Fails = { "vale-dead", "vale-leaves", "rivals-reach-vale", "sealed", "enforcer-down", "entry-seen-guard", "entry-seen-camera", "entry-seen-patrol", "abduction-seen-camera", "driver-shot", "engine-error" };
    static readonly string[] Kinds = { "camera", "guard", "patrol", "officer", "robot", "rival", "person", "vale", "op", "car", "desk", "heard" };

    /// <summary>The cause's shape, whatever the ending: it matches the result, it is at the last tick, a single culprit is named and placed
    /// first, a seal names its origin, and its JSON is what the outcome writes.</summary>
    static void WellFormed(Game g, string what)
    {
        if (g.Result == Outcome.Success || g.Result == Outcome.None) { Assert.True(g.Cause == null, $"{what}: no cause without a failure"); return; }
        var c = g.Cause; Assert.True(c != null, $"{what}: {g.Result} {g.FailCode} carries a cause");
        Assert.Equal(g.Result == Outcome.Withdrawn ? "withdraw" : g.FailCode, c!.Fail);
        Assert.Contains(c.Fail, Fails.Append("withdraw"));
        Assert.Equal(g.Tick, c.Tick);
        Assert.NotEqual("unresolved", c.Why);
        foreach (var b in c.Culprits) { Assert.Contains(b.Kind, Kinds); Assert.False(string.IsNullOrEmpty(b.Id), $"{what}: a culprit with no id"); }
        foreach (var o in c.Ops) Assert.Contains(o, new[] { "scout", "fixer", "tech", "enforcer" });
        if (c.Single) { Assert.NotEmpty(c.Culprits); Assert.True(c.Culprits[0].Placed, $"{what}: the culprit to orbit is placed ({c.Culprits[0]})"); }
        if (c.Fail == "sealed") { Assert.False(c.Single); Assert.True(c.Origin != null && c.Origin.Tick < c.Tick && c.Origin.Report.Length > 0, $"{what}: the seal names what started it"); }
        var json = Report.Json(what, g, new List<(bool, string)>(), true);
        Assert.Contains("\"cause\":" + c.Json(), json);
    }
